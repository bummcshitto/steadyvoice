cmake_minimum_required(VERSION 3.16)
project(SteadyVoice VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(SteadyVoice MODULE src/plugin.cpp src/editor.cpp)
set_target_properties(SteadyVoice PROPERTIES PREFIX "")
if(WIN32)
  enable_language(RC)
  target_sources(SteadyVoice PRIVATE src/SteadyVoice.rc)
  target_link_libraries(SteadyVoice PRIVATE user32 gdi32)
endif()
if(MSVC)
  set_property(TARGET SteadyVoice PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(SteadyVoice PRIVATE /O2 /W4 /fp:precise)
else()
  target_compile_options(SteadyVoice PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-math-errno)
  if(MINGW)
    target_link_options(SteadyVoice PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(adaptive_tests tests/adaptive_tests.cpp)
  add_test(NAME adaptive_behavior COMMAND adaptive_tests)
  add_executable(dsp_tests tests/dsp_tests.cpp)
  add_test(NAME dsp_behavior COMMAND dsp_tests)
  add_executable(loudness_tests tests/loudness_tests.cpp)
  add_test(NAME loud_transitions COMMAND loudness_tests)
  add_executable(guard_quality_tests tests/guard_quality_tests.cpp)
  add_test(NAME adaptive_guard_quality COMMAND guard_quality_tests)
  add_executable(true_peak_tests tests/true_peak_tests.cpp)
  add_test(NAME peak_detector_quality COMMAND true_peak_tests)
  add_executable(peak_guard_tests tests/peak_guard_tests.cpp)
  add_test(NAME oversampled_peak_guard COMMAND peak_guard_tests)
  add_executable(host_tests tests/host_tests.cpp)
  if(WIN32)
    target_link_libraries(host_tests PRIVATE user32 gdi32)
  else()
    target_link_libraries(host_tests PRIVATE dl pthread)
    target_link_options(host_tests PRIVATE -rdynamic)
  endif()
  add_test(NAME plugin_host COMMAND host_tests $<TARGET_FILE:SteadyVoice>)
endif()
